# ============================================
# ASTRO AIRFLOW COMMANDS CHEAT SHEET
# ============================================

astro dev init
# Creates a new Astro Airflow project.
# Run this only once when starting a new project.

astro dev start
# Starts your local Airflow environment.
# Builds the Docker image if needed.
# Starts the Scheduler, Webserver, Triggerer and Metadata Database.
# Use when starting work or after running 'astro dev stop' or 'astro dev kill'.

astro dev stop
# Stops all Airflow containers.
# Keeps the containers so they can be started again quickly.
# Use when you're finished working for the day.

astro dev restart
# Restarts the Airflow environment.
# Equivalent to stopping and starting the services.
# Use after changing requirements.txt, Dockerfile, or environment variables.

astro dev kill
# Stops and removes all Airflow containers.
# Use when Docker is stuck or you need a clean environment.
# Your project files are NOT deleted.

astro dev logs
# Displays logs from all Airflow services.

astro dev logs scheduler
# Displays only the Scheduler logs.
# Best place to debug DAG import errors and scheduling issues.

astro dev logs webserver
# Displays only the Webserver logs.
# Useful for debugging UI-related issues.

astro dev bash
# Opens a terminal inside the Airflow container.
# Use to run Airflow CLI commands or inspect the environment.

exit
# Exits the Airflow container shell and returns to your local terminal.

astro dev parse
# Parses all DAGs and checks for import/syntax errors without starting Airflow.
# Great for validating DAGs before running them.

astro deploy
# Deploys your project to an Astronomer-hosted environment.
# Not used for local development.

astro login
# Authenticates your Astro CLI with an Astronomer account.

astro logout
# Logs out of your Astronomer account.

astro version
# Displays the installed Astro CLI version.
